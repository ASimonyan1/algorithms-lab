cmake_minimum_required(VERSION 3.16)
project(AlgorithmsLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(algolab INTERFACE)
target_include_directories(algolab INTERFACE include)
if(MSVC)
  target_compile_options(algolab INTERFACE /W4 /WX)
else()
  target_compile_options(algolab INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(ENABLE_SANITIZERS)
    target_compile_options(algolab INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(algolab INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_executable(demo examples/demo.cpp)
target_link_libraries(demo PRIVATE algolab)
enable_testing()
add_executable(algolab_tests tests/test_algolab.cpp)
target_link_libraries(algolab_tests PRIVATE algolab)
add_test(NAME algolab_tests COMMAND algolab_tests)
